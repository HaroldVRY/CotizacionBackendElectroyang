-- ============ INICIO SECCIÓN 1 ============ --
-- Crear extensión para UUID si es necesario
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Función para actualizar timestamps
CREATE OR REPLACE FUNCTION update_timestamp()
RETURNS TRIGGER AS $$
BEGIN
   NEW.updated_at = CURRENT_TIMESTAMP;
   RETURN NEW;
END;
$$ LANGUAGE plpgsql;
-- ============ FIN SECCIÓN 1 ============ --
-- ============ INICIO SECCIÓN 2 ============ --
-- Tabla de usuarios
CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    rol VARCHAR(20) CHECK (rol IN ('admin', 'usuario')) DEFAULT 'usuario',
    activo BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de clientes
CREATE TABLE cliente (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    ruc VARCHAR(20),
    direccion TEXT,
    telefono VARCHAR(20),
    email VARCHAR(255),
    contacto VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla de servicios
CREATE TABLE servicio (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    descripcion TEXT,
    precio DECIMAL(10,2),
    unidad VARCHAR(50),
    activo BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
-- ============ FIN SECCIÓN 2 ============ --
-- ============ INICIO SECCIÓN 3 ============ --
-- Tabla de cotizaciones
CREATE TABLE cotizacion (
    id SERIAL PRIMARY KEY,
    numero VARCHAR(50) UNIQUE NOT NULL,
    fecha DATE NOT NULL,
    cliente_id INT NOT NULL,
    usuario_id INT NOT NULL,
    receptor VARCHAR(255),
    observaciones TEXT,
    tiempo_entrega VARCHAR(255) NOT NULL,
    forma_pago VARCHAR(255) NOT NULL,
    estado VARCHAR(20) CHECK (estado IN ('borrador', 'enviada', 'aprobada', 'rechazada')) DEFAULT 'borrador',
    subtotal DECIMAL(10,2) NOT NULL DEFAULT 0,
    igv DECIMAL(10,2) NOT NULL DEFAULT 0,
    total DECIMAL(10,2) NOT NULL DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES cliente(id) ON DELETE RESTRICT,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id) ON DELETE RESTRICT
);

-- Tabla de detalle de cotizaciones
CREATE TABLE detalle_cotizacion (
    id SERIAL PRIMARY KEY,
    cotizacion_id INT NOT NULL,
    servicio_id INT,
    numero_item INT NOT NULL,
    cantidad DECIMAL(10,3) NOT NULL,
    descripcion TEXT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    total DECIMAL(10,2) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cotizacion_id) REFERENCES cotizacion(id) ON DELETE CASCADE,
    FOREIGN KEY (servicio_id) REFERENCES servicio(id) ON DELETE SET NULL
);
-- ============ FIN SECCIÓN 3 ============ --
-- ============ INICIO SECCIÓN 4 ============ --
-- Triggers para actualización automática de timestamps
CREATE TRIGGER update_usuario_timestamp
BEFORE UPDATE ON usuario
FOR EACH ROW EXECUTE FUNCTION update_timestamp();

CREATE TRIGGER update_cliente_timestamp
BEFORE UPDATE ON cliente
FOR EACH ROW EXECUTE FUNCTION update_timestamp();

CREATE TRIGGER update_servicio_timestamp
BEFORE UPDATE ON servicio
FOR EACH ROW EXECUTE FUNCTION update_timestamp();

CREATE TRIGGER update_cotizacion_timestamp
BEFORE UPDATE ON cotizacion
FOR EACH ROW EXECUTE FUNCTION update_timestamp();

CREATE TRIGGER update_detalle_timestamp
BEFORE UPDATE ON detalle_cotizacion
FOR EACH ROW EXECUTE FUNCTION update_timestamp();

-- Función para generar números de cotización automáticos
CREATE OR REPLACE FUNCTION generar_numero_cotizacion()
RETURNS TRIGGER AS $$
DECLARE
    ultimo_numero INT;
    anio_actual INT;
BEGIN
    anio_actual := EXTRACT(YEAR FROM CURRENT_DATE);
    
    SELECT COALESCE(
        MAX(CAST(SUBSTRING(numero FROM 10 FOR 4) AS INT)), 
        0
    ) INTO ultimo_numero
    FROM cotizacion
    WHERE numero LIKE 'COT-' || anio_actual || '-%';
    
    NEW.numero := 'COT-' || anio_actual || '-' || LPAD((ultimo_numero + 1)::TEXT, 4, '0');
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger para asignar número de cotización automático
CREATE TRIGGER trigger_generar_numero_cotizacion
BEFORE INSERT ON cotizacion
FOR EACH ROW
WHEN (NEW.numero IS NULL)
EXECUTE FUNCTION generar_numero_cotizacion();
